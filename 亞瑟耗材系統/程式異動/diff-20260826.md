## apps/api/src/consumable/forecast-labels.ts
```typescript
/**
 * 預測看板顯示標籤解析（2026/08/25 缺陷修）。
 *
 * 背景：預測來源是「歷史 ∪ baselines」。歷史來自 `CS_Order`（下單當下就固化 ProjectCode、永不改），
 * 而 `CS_SiteItem` / `CS_Site` 的 ProjectCode 是「今日生效主約」快照、會隨掛約異動漂移。兩者分岔時，
 * 「有歷史但已不在現行清單」的站×品項就查不到 meta。
 *
 * 修前行為：meta miss 直接把**代碼**當名稱吐出去（站名顯示 `ST-GTC-004`、品名顯示 `C01A-001`、單位空白），
 * 而且那份假名稱還會被寫進 `CS_Forecast` 快照永久固化。
 *
 * 本 helper 的規則：
 *   1. 有名稱 → 用名稱。
 *   2. 沒有名稱（null/空白），或名稱剛好等於代碼（＝被舊 fallback 汙染的快照）→ 標示成 `代碼(已停用)`。
 *   3. 連代碼都沒有 → 空字串。
 *
 * 重點是**不可靜默吐裸代碼**：裸代碼在畫面上看起來像正常名稱，使用者無從分辨是「主檔查無」還是「命名怪」。
 */

/** 主檔查無名稱時附加的標示（前端不需特判，直接顯示即可） */
export const MASTER_MISSING_SUFFIX = '(已停用)';

/**
 * 產生顯示用標籤。
 *
 * @param name 主檔／快照拿到的名稱（可能 null、空白，或被舊 fallback 汙染成代碼本身）
 * @param code 對應的代碼（站碼／品項碼）
 */
export function displayLabel(
  name: string | null | undefined,
  code: string | null | undefined,
): string {
  const n = (name ?? '').trim();
  const c = (code ?? '').trim();
  if (n && n !== c) return n;
  return c ? `${c}${MASTER_MISSING_SUFFIX}` : n;
}
```

## apps/api/src/consumable/supplier-priority.ts
```typescript
/**
 * 商品主檔的「供應商優先序」與請購轉採購的自動帶入（季河 Kevin 2026-08-26）。
 *
 * 業主原話：「在商品主檔設定有哪些供應商可以供應並且可以排順序，
 * 然後請購單轉採購單的時候直接按最優先的供應商帶路轉成採購單」。
 *
 * 為什麼要有這支獨立模組：
 *   帶入規則牽涉四種來源（人工已指定／主檔順序／上次買的／能供清單），
 *   而且每一種都要能在畫面上說清楚「為什麼是這家」。把它留在 5,000 行的
 *   consumable.service 裡就只能靠連 DB 的整合測試驗，規則一改沒人敢動。
 *   抽成純函式後，四條規則各自可測，畫面文案也有單一出處。
 *
 * 三件不能忘的事：
 *   1. **人工指定最大**。CS_OrderItem.SupplierCode 一旦有值就代表現場主管或採購
 *      已經挑過了，自動帶入絕不覆蓋 —— 這是「預設值可改」的前提，
 *      改完重新整理還被系統蓋回去，等於沒得改。
 *   2. **停用或沒報價就跳過**。第一順位被停用、缺貨或還沒報價都是常態，
 *      跳到下一順位，並把「跳過誰、為什麼」一起回給畫面（兩種原因分開講）。
 *   3. **全部都不能帶就留空**。硬帶一家沒報價的供應商，採購看到欄位有值
 *      就不會再確認，最後是用錯價格發單。寧可空著讓人選。
 *
 * 相容性：主檔順序沒設過的品項（Priority 全空）維持既有行為
 * —— 供應商對齊批 3a 的「上次買的」，不會因為這次改動而變動既有畫面。
 */

// ═══════════════════════════════════════════════════════════════════
// 定案一（Kevin 2026-08-26）：報價認定維持寬鬆，但要標示來源
//
//   業主原話：「先用寬的方式到時候補好資料再改成一定要有報價表才能夠使用」。
//   所以現在三個來源任一都算，畫面上一律標出這個價是哪來的，
//   採購看得出「物價表」跟「上次買的」的可信度不一樣。
//
//   **日後要收緊成「只認物價表」時，只改下面 QUOTE_MODE 這一個常數即可**
//   （改成 'strict'）。判定散在各處的話，收緊那天就得逐處找、逐處漏。
//   收緊後：只有 CS_PriceList 現價算報價，僅有參考價／上次採購價的組合
//   會被視為「無現行報價」而跳過，畫面文案自動跟著變，不需要改別的地方。
// ═══════════════════════════════════════════════════════════════════

/** 報價來源，可信度由高到低 */
export type QuoteSource = '物價表' | '參考價' | '上次買的';

export type QuoteMode = 'loose' | 'strict';

/** 現行寬鬆模式。資料補齊後改成 'strict' 就收緊成「一定要有物價表」 */
export const QUOTE_MODE: QuoteMode = 'loose';

export const QUOTE_SOURCES_BY_MODE: Record<QuoteMode, readonly QuoteSource[]> = {
  loose: ['物價表', '參考價', '上次買的'],
  strict: ['物價表'],
};

/** 目前這個模式下「算數」的報價來源 */
export function acceptedQuoteSources(mode: QuoteMode = QUOTE_MODE): readonly QuoteSource[] {
  return QUOTE_SOURCES_BY_MODE[mode];
}

// ═══════════════════════════════════════════════════════════════════
// 定案三（Kevin 2026-08-26）：停用某家供應商
//
//   2026-08-26 與客戶確認：舊系統與現行新系統都沒有「對這個品項停用某家供應商」
//   的設定，所以本次新增，位置就放在商品主檔的「供應商順序」畫面
//   —— 停用跟排序是同一件事的兩面（這家要不要用、排第幾），放同一頁才不用跑兩處。
//
//   存哪裡：沿用 CS_ItemSupplier 既有的 Status 欄，不加欄位、不用 migration。
//     'C' ＝ 移除能供應關係（既有語意，軟刪，各處查詢一律 `Status <> 'C'` 就看不到）
//     'N' ＝ 停用（本次新增，**可逆**，仍留在能供清單裡，隨時可以開回來）
//     其餘（'Y' / NULL）＝ 啟用
//   選 Status 而不是新增 IsEnabled 欄的理由：CS_ItemSupplier 的啟用狀態本來就由
//   Status 表達，多開一個布林欄會出現「Status='C' 但 IsEnabled=true」這種說不清的組合。
//
//   停用只擋「自動帶入」，不改能供清單本身：採購／現場主管仍可手動選那一家
//   （人工指定永遠最大，見上面第 1 點）。要徹底不能選，走既有的「移除能供應品項」。
// ═══════════════════════════════════════════════════════════════════

/** CS_ItemSupplier.Status ＝ 停用（可逆；'C' 是既有的軟刪，語意不同） */
export const ITEM_SUPPLIER_DISABLED = 'N';

/** CS_ItemSupplier.Status ＝ 啟用 */
export const ITEM_SUPPLIER_ENABLED = 'Y';

/** CS_ItemSupplier 一列（只取排序需要的欄位；Priority 欄早已存在，不需 migration） */
export interface ItemSupplierLink {
  SupplierCode: string;
  /** 多源優先序，越小越優先。null = 主檔還沒排過 */
  Priority?: number | null;
  /** 舊資料的「主源」旗標。沒排順序時當次要依據 */
  IsPrimary?: boolean | null;
  /** 建檔順序，最後的平手裁決 */
  Seq?: number | null;
  /** 'N' = 停用（可逆）；'C' 的列根本不會查出來；其餘/NULL = 啟用 */
  Status?: string | null;
}

export interface RankedSupplier {
  supplierCode: string;
  /** 主檔設定的順序值；null = 未設 */
  priority: number | null;
  /** 排序後的實際名次（1 起算），畫面顯示用 */
  rank: number;
  /** false = 這個品項停用了這家（名次仍保留，見 rankItemSuppliers 註解） */
  enabled: boolean;
}

/** 帶入依據。空字串 = 沒帶（留給人工指定） */
export type PickBasis = '已指定' | '主檔優先序' | '上次買的' | '能供首家' | '';

/** 跳過的兩種原因，畫面要分開顯示，不能混成一句 */
export type SkipKind = 'disabled' | 'no-quote';

export interface SkippedSupplier {
  supplierCode: string;
  rank: number;
  /** 機器判讀用：'disabled' = 被停用；'no-quote' = 沒有現行報價 */
  kind: SkipKind;
  /** 人看的原因文字 */
  reason: string;
}

export interface PickResult {
  supplierCode: string;
  basis: PickBasis;
  /** 帶入這家時採用的價來源；沒帶入＝空字串 */
  quoteSource: QuoteSource | '';
  /** 畫面上「為什麼建議這家」的完整說明 */
  reason: string;
  /** 被跳過的供應商（依名次），畫面用 tooltip 列出來 */
  skipped: SkippedSupplier[];
}

export interface PickSupplierInput {
  /** 能供該品項的供應商，已排序（rankItemSuppliers 的輸出） */
  ranked: RankedSupplier[];
  /** 該供應商對該品項的報價來源；空字串 = 這個模式下不算有報價 */
  quoteOf: (supplierCode: string) => QuoteSource | '';
  /** 已由人指定的供應商（CS_OrderItem.SupplierCode）。有值就不覆蓋 */
  manualPick?: string | null;
  /** 該品項最近一筆已發單的供應商（供應商對齊批 3a 的既有規則） */
  lastPurchased?: string | null;
  /** 供應商主檔整家停用（CS_Supplier.IsActive=false）。與品項層停用原因分開顯示 */
  supplierDisabled?: (supplierCode: string) => boolean;
}

const NO_PRIORITY = Number.MAX_SAFE_INTEGER;

const clean = (v: string | null | undefined): string => (v ?? '').trim();

/** Status 是不是「停用」（大小寫都收，舊資料手寫過 'n' 也認） */
export function isDisabledStatus(status: string | null | undefined): boolean {
  return clean(status).toUpperCase() === ITEM_SUPPLIER_DISABLED;
}

/**
 * 把能供應清單排成明確的順序。
 *
 * 排序鍵依序是：主檔順序 → 主源旗標 → 建檔序 → 代碼字典序。
 * 最後一層用代碼是刻意的：全部欄位都沒設時也要有穩定結果，
 * 否則同一批資料每次查出來的「第一家」會跟著 DB 掃描順序跑。
 *
 * **停用的那幾家不移到最後、名次照舊**：停用是暫時的，開回來時名次要跟停用前一樣，
 * 排到最後會讓「停用→啟用」變成一次隱形的重新排序，採購排好的順序會莫名其妙被洗掉。
 * 畫面上灰顯＋標「已停用」就足夠分辨，帶入時才真的跳過。
 */
export function rankItemSuppliers(links: readonly ItemSupplierLink[]): RankedSupplier[] {
  const seen = new Set<string>();
  const uniq: ItemSupplierLink[] = [];
  for (const l of links) {
    const code = clean(l.SupplierCode);
    if (!code || seen.has(code)) continue;
    seen.add(code);
    uniq.push({ ...l, SupplierCode: code });
  }
  const keyOf = (l: ItemSupplierLink): [number, number, number, string] => [
    l.Priority == null ? NO_PRIORITY : Number(l.Priority),
    l.IsPrimary === true ? 0 : 1,
    l.Seq == null ? NO_PRIORITY : Number(l.Seq),
    clean(l.SupplierCode),
  ];
  return [...uniq]
    .sort((a, b) => {
      const ka = keyOf(a);
      const kb = keyOf(b);
      for (let i = 0; i < 3; i++) {
        const d = (ka[i] as number) - (kb[i] as number);
        if (d !== 0) return d;
      }
      return (ka[3] as string).localeCompare(kb[3] as string);
    })
    .map((l, i) => ({
      supplierCode: clean(l.SupplierCode),
      priority: l.Priority == null ? null : Number(l.Priority),
      rank: i + 1,
      enabled: !isDisabledStatus(l.Status),
    }));
}

/** 主檔有沒有真的排過順序。全空 = 沒排過，維持既有的「上次買的」規則 */
export function hasConfiguredPriority(ranked: readonly RankedSupplier[]): boolean {
  return ranked.some((r) => r.priority != null);
}

const rankLabel = (r: RankedSupplier): string =>
  r.priority == null ? '未設順序' : `第 ${r.priority} 順位`;

/** 「2 家已停用、1 家無現行報價」—— 兩種原因分開講，採購才知道要去開哪個開關 */
function skipSummary(skipped: readonly SkippedSupplier[]): string {
  const disabled = skipped.filter((s) => s.kind === 'disabled').length;
  const noQuote = skipped.filter((s) => s.kind === 'no-quote').length;
  const parts: string[] = [];
  if (disabled) parts.push(`${disabled} 家已停用`);
  if (noQuote) parts.push(`${noQuote} 家無現行報價`);
  return parts.join('、');
}

/**
 * 決定請購轉採購要帶哪一家。
 *
 * 規則優先序（由高到低）：
 *   1. 人工已指定 → 原封不動回傳，不覆蓋
 *   2. 主檔排過順序 → 依序找第一家「已啟用且有現行報價」的，其餘跳過並記錄原因
 *   3. 主檔沒排過順序 → 沿用既有的「上次買的」（且該家仍在能供清單、未停用、仍有報價）
 *   4. 再不然 → 能供清單第一家（同樣要求已啟用且有報價）
 *   5. 都沒有 → 留空，畫面提示請人工指定
 */
export function pickSupplier(input: PickSupplierInput): PickResult {
  const { ranked, quoteOf } = input;
  const supplierDisabled = input.supplierDisabled ?? (() => false);
  const manual = clean(input.manualPick);
  if (manual) {
    return {
      supplierCode: manual,
      basis: '已指定',
      quoteSource: quoteOf(manual) || '',
      reason: '已由人工指定，系統不覆蓋',
      skipped: [],
    };
  }
  if (!ranked.length) {
    return {
      supplierCode: '',
      basis: '',
      quoteSource: '',
      reason: '商品主檔尚未設定可供應此品項的供應商，請先到商品主檔設定',
      skipped: [],
    };
  }

  const skipped: SkippedSupplier[] = [];
  /** 這家能不能自動帶入；不能就記錄原因並回 null */
  const usable = (r: RankedSupplier): QuoteSource | '' | null => {
    if (!r.enabled) {
      skipped.push({
        supplierCode: r.supplierCode,
        rank: r.rank,
        kind: 'disabled',
        reason: '已停用',
      });
      return null;
    }
    if (supplierDisabled(r.supplierCode)) {
      skipped.push({
        supplierCode: r.supplierCode,
        rank: r.rank,
        kind: 'disabled',
        reason: '供應商已停用',
      });
      return null;
    }
    const src = quoteOf(r.supplierCode);
    if (!src) {
      skipped.push({
        supplierCode: r.supplierCode,
        rank: r.rank,
        kind: 'no-quote',
        reason: '無現行報價',
      });
      return null;
    }
    return src;
  };
  const firstUsable = (): { hit: RankedSupplier; source: QuoteSource } | undefined => {
    for (const r of ranked) {
      const src = usable(r);
      if (src) return { hit: r, source: src };
    }
    return undefined;
  };

  if (hasConfiguredPriority(ranked)) {
    const found = firstUsable();
    if (found) {
      const detail = skipped.length ? `；已跳過 ${skipSummary(skipped)}` : '';
      return {
        supplierCode: found.hit.supplierCode,
        basis: '主檔優先序',
        quoteSource: found.source,
        reason: `商品主檔${rankLabel(found.hit)}，且有現行報價（${found.source}）${detail}`,
        skipped,
      };
    }
    return {
      supplierCode: '',
      basis: '',
      quoteSource: '',
      reason: `主檔 ${ranked.length} 家供應商都不能自動帶入（${skipSummary(skipped)}），請人工指定`,
      skipped,
    };
  }

  // 主檔沒排過順序 —— 維持供應商對齊批 3a 的既有行為，不因這次改動而變
  const last = clean(input.lastPurchased);
  const lastRow = last ? ranked.find((r) => r.supplierCode === last) : undefined;
  if (lastRow && lastRow.enabled && !supplierDisabled(last)) {
    const src = quoteOf(last);
    if (src) {
      return {
        supplierCode: last,
        basis: '上次買的',
        quoteSource: src,
        reason: `商品主檔未排供應商順序，沿用此品項上次採購的供應商（價來源：${src}）`,
        skipped: [],
      };
    }
  }
  const found = firstUsable();
  if (found) {
    const detail = skipped.length ? `（已跳過 ${skipSummary(skipped)}）` : '';
    return {
      supplierCode: found.hit.supplierCode,
      basis: '能供首家',
      quoteSource: found.source,
      reason: `商品主檔未排供應商順序，取能供應清單中第一家可用者（價來源：${found.source}）${detail}`,
      skipped,
    };
  }
  return {
    supplierCode: '',
    basis: '',
    quoteSource: '',
    reason: `能供應此品項的 ${ranked.length} 家供應商都不能自動帶入（${skipSummary(skipped)}），請人工指定`,
    skipped,
  };
}

/**
 * 把「哪些供應商有報價、報價是哪來的」收斂成一個查詢函式。
 *
 * 報價的認定刻意放寬到三個來源，因為業主端這三個地方都算「談好的價」：
 *   - 物價表現價 CS_PriceList.IsCurrent
 *   - 基本參考價 CS_ItemSupplier.ReferencePrice（亞瑟自己設的）
 *   - 最近採購價 CS_OrderItem.Price（供應商對齊批 3b 認定的現價）
 * 只認其中一種，會讓另外兩種資料齊全的品項被誤判成「沒報價」而整批留空。
 *
 * 回傳的是**來源**不是布林：畫面要標「這個價是哪來的」（定案一）。
 * mode 預設吃 QUOTE_MODE；收緊那天改 QUOTE_MODE 一處，這裡與所有畫面同步生效。
 */
export function quoteLookup(
  quotedSources: ReadonlyMap<string, QuoteSource>,
  mode: QuoteMode = QUOTE_MODE,
): (itemCode: string, supplierCode: string) => QuoteSource | '' {
  const accepted = new Set<QuoteSource>(acceptedQuoteSources(mode));
  return (itemCode, supplierCode) => {
    const src = quotedSources.get(`${itemCode}|${supplierCode}`);
    return src && accepted.has(src) ? src : '';
  };
}
```
